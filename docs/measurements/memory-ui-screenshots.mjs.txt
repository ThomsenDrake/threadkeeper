import assert from 'node:assert/strict';
import {createRequire} from 'node:module';
import {readFile,writeFile} from 'node:fs/promises';
const require=createRequire('/opt/codex/runtimes/codex-primary-runtime/dependencies/node/node_modules/playwright/index.js');
const {chromium}=require('playwright');
const browser=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox']});
const page=await browser.newPage({viewport:{width:1512,height:1100}});
const errors=[];page.on('pageerror',error=>errors.push(error.message));
const fixture=JSON.parse(await readFile('/tmp/threadkeeper-memory-ui-fixture.json','utf8'));
const panel=()=>page.getByRole('dialog',{name:'Memory details',exact:true});
const shots=[];
async function shot(name){const path=`/workspace/threadkeeper/docs/images/${name}.png`;if(name.startsWith('memory-')) await panel().screenshot({path}); else if(name==='connections') await page.locator('.connections-panel').screenshot({path}); else if(name==='portability') await page.locator('.portability-grid').screenshot({path}); else await page.screenshot({path,fullPage:false});shots.push(path);}
try{
 await page.goto('http://127.0.0.1:3000');
 await page.getByLabel('Email',{exact:true}).fill('maya@example.invalid');
 await page.getByLabel('Password',{exact:true}).fill('threadkeeper-demo-password');
 await page.getByRole('button',{name:'Sign in',exact:true}).click();
 await page.getByRole('heading',{name:'What Threadkeeper holds for you',exact:true}).waitFor();
 await page.locator('.principal-plate').waitFor();
 assert.equal(await page.getByRole('button',{name:/Needs review|^Confirm$|Edit and confirm|^Dismiss$/}).count(),0);
 await shot('memories');
 await page.setViewportSize({width:390,height:844});
 assert.ok(await page.evaluate(()=>document.documentElement.scrollWidth<=390));
 await page.evaluate(()=>window.scrollTo(0,500));
 await shot('memories-phone');
 await page.setViewportSize({width:1512,height:1700});
 await page.evaluate(()=>window.scrollTo(0,0));
 await page.getByRole('button',{name:/Browse all memories/}).click();
 await page.locator(`.memory-row[data-memory-id="${fixture.ids.beta}"]`).click();
 await panel().locator('.detail-statement').waitFor();
 assert.ok((await panel().innerText()).includes('Corrected by you'));
 assert.ok((await panel().innerText()).includes('Kept in history, no longer delivered'));
 assert.ok((await panel().textContent()).includes('Preserved as captured'));
 assert.equal(await panel().locator('.junction-seal .seal-centre path[d="M38 50.5 L46.5 59 L63 41"]').count(),0,'Provenance has a check seal');
 await shot('memory-detail');
 await panel().getByRole('button',{name:'Close memory details',exact:true}).click();
 await page.locator(`.memory-row[data-memory-id="${fixture.ids.digest}"]`).click();
 await panel().locator('.detail-statement').waitFor();
 assert.ok((await panel().innerText()).includes('From an assistant'));
 assert.ok((await panel().innerText()).includes('Available to two agents'));
 assert.equal(await panel().getByRole('button',{name:'Correct',exact:true}).count(),1);
 assert.equal(await panel().getByRole('button',{name:'Forget',exact:true}).count(),1);
 assert.doesNotMatch(await panel().innerText(),/unaccepted|not accepted|have not accepted|needs review/i);
 await shot('memory-source');
 await panel().getByRole('button',{name:'Close memory details',exact:true}).click();
 await page.setViewportSize({width:1512,height:1100});
 await page.getByRole('navigation',{name:'Profile controls'}).getByRole('button',{name:'Connections',exact:true}).click();
 await page.getByRole('heading',{name:'Client credentials.',exact:true}).waitFor();
 await page.getByRole('heading',{name:'Writing assistant',exact:true}).waitFor();
 await shot('connections');
 await page.getByRole('navigation',{name:'Profile controls'}).getByRole('button',{name:'Captures',exact:true}).click();
 await page.getByRole('heading',{name:'Follow your captures.',exact:true}).waitFor();
 await page.locator(`[data-capture-id="${fixture.completed}"]`).waitFor();
 const completed=page.locator(`[data-capture-id="${fixture.completed}"]`);
 assert.ok((await completed.innerText()).includes('1 memory saved'));
 assert.doesNotMatch(await completed.innerText(),/\baccepted\b/);
 assert.equal(await page.getByRole('button',{name:'Retry extraction',exact:true}).count(),1);
 await page.setViewportSize({width:1512,height:2050});
 await page.evaluate(()=>window.scrollTo(0,0));
 await shot('captures');
 await page.getByRole('navigation',{name:'Profile controls'}).getByRole('button',{name:'Import & export',exact:true}).click();
 await page.getByRole('heading',{name:'Memory without lock-in.',exact:true}).waitFor();
 await shot('portability');
 assert.deepEqual(errors,[]);
 await writeFile('/tmp/threadkeeper-memory-ui-screenshots.json',JSON.stringify({mode:'Synthetic disposable PGlite; actual production React/Express profile; Chromium',screenshots:shots,browser_errors:errors},null,2)+'\n');
 console.log(`Captured ${shots.length} production UI screenshots; no browser errors.`);
}finally{await browser.close();}
