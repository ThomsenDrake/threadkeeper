import {createHash} from 'node:crypto';
import {execFileSync} from 'node:child_process';
import {readFile,readdir} from 'node:fs/promises';
import {join,relative} from 'node:path';
export const root='/workspace/threadkeeper-main-review';
const sourcePaths=['apps/api','apps/web/src','apps/web/package.json','apps/web/tsconfig.json','apps/web/vite.config.ts','packages/core','packages/contracts','packages/providers','deploy/migrations','tests/helpers.ts','package.json','pnpm-lock.yaml','pnpm-workspace.yaml','tsconfig.json'];
const sha=data=>createHash('sha256').update(data).digest('hex');
async function files(path){const entries=await readdir(path,{withFileTypes:true});return(await Promise.all(entries.map(entry=>entry.isDirectory()?files(join(path,entry.name)):[join(path,entry.name)]))).flat();}
export async function captureSource(){
 const source_commit=execFileSync('git',['rev-parse','HEAD'],{cwd:root,encoding:'utf8'}).trim();
 const paths=execFileSync('git',['ls-files','--',...sourcePaths],{cwd:root,encoding:'utf8'}).trim().split('\n').filter(Boolean).sort();
 const source_sha256=Object.fromEntries(await Promise.all(paths.map(async path=>[path,sha(await readFile(join(root,path)))])));
 const build_sha256=Object.fromEntries(await Promise.all((await files(join(root,'apps/web/dist'))).sort().map(async path=>[relative(root,path),sha(await readFile(path))])));
 return {source_commit,source_patch_sha256:sha(execFileSync('git',['diff','--binary','HEAD','--',...sourcePaths],{cwd:root})),app_patch_sha256:sha(execFileSync('git',['diff','--binary','HEAD','--','apps/web/src/App.tsx'],{cwd:root})),source_sha256,build_sha256};
}
