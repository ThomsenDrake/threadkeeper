import assert from 'node:assert/strict';
import {writeFile} from 'node:fs/promises';
import {captureSource,root} from './source.mjs';
const source=await captureSource();
assert.equal(source.source_commit,process.env.TK_UI_BASE_COMMIT,'Unexpected source base');
assert.equal(source.source_sha256['apps/web/src/App.tsx'],process.env.TK_UI_APP_SHA256,'Unexpected profile patch');
const {createTestDatabase}=await import(root+'/tests/helpers.ts');
const {bootstrap}=await import(root+'/apps/api/src/auth.ts');
const {createApp}=await import(root+'/apps/api/src/app.ts');
const database=await createTestDatabase();
await bootstrap(database.db,'demo@example.invalid','threadkeeper-demo-password');
const ownerId=(await database.db.query('SELECT id FROM tk_users')).rows[0].id;
const origin='http://127.0.0.1:38118';
const {app,store}=createApp(database.db,{origin,staticDir:root+'/apps/web/dist'});
const profile={ownerId,clientId:'profile',permissions:['*'],projects:null};
const client={...profile,clientId:'synthetic-date-browser-client',permissions:['read','capture']};
const originalDate='2026-10-03T12:34:56.789Z';
const seeds={
 activeClear:{statement:'Synthetic Alder reviews use short paragraphs.',candidate:false},
 candidateClear:{statement:'Synthetic Birch notes may use concise tables.',candidate:true},
 activePrecise:{statement:'Synthetic Cedar reviews use numbered steps.',candidate:false,effective_at:'2027-11-07T06:30:56.789Z'},
 candidatePrecise:{statement:'Synthetic Dogwood notes may use brief headings.',candidate:true,effective_at:'2027-11-07T06:30:56.789Z'},
 activeChange:{statement:'Synthetic Elm updates use bullet lists.',candidate:false},
 candidateChange:{statement:'Synthetic Fern notes may use plain language.',candidate:true},
 confirm:{statement:'Synthetic Grove notes may use short examples.',candidate:true},
 dismiss:{statement:'Synthetic Hazel notes may use weekly summaries.',candidate:true},
 race:{statement:'Synthetic Iris updates use quarterly summaries.',candidate:false},
};
try{
 for(const [name,seed] of Object.entries(seeds)){
  const eventId='synthetic-date-'+name;
  const candidate={statement:seed.statement,kind:'preference',source_event_id:eventId,quote:seed.statement,origin:seed.candidate?'inferred':'user_explicit',effective_at:seed.effective_at??originalDate};
  const receipt=await store.capture(client,{idempotency_key:eventId,project_id:'synthetic-date-ui',subject:'self',events:[{id:eventId,text:seed.statement,author_role:'user',origin:'user_explicit'}],...(seed.candidate?{}:{explicit_memories:[candidate]})});
  if(seed.candidate){const result=await store.processJob({extract:async()=>({memories:[candidate],model:'synthetic-date-browser-fixture'})});assert.equal(result?.accepted,1);}
  const status=await store.captureStatus(profile,receipt.capture_id);seed.id=status.memory_ids[0];seed.before=await store.detail(profile,seed.id);
 }
 assert.deepEqual(await captureSource(),source,'Runtime/build files changed during setup');
 await writeFile('/tmp/threadkeeper-date-ui/fixture.json',JSON.stringify({source,origin,originalDate,seeds},null,2));
 const server=app.listen(38118,'127.0.0.1',()=>console.log('READY disposable synthetic date profile'));
 let closing=false;
 for(const signal of ['SIGINT','SIGTERM'])process.on(signal,()=>{if(closing)return;closing=true;server.close(async()=>{await database.close();console.log('CLEANUP complete');process.exit(0);});});
}catch(error){await database.close();throw error;}
